let current=0;
let answers={};

function loadQ(){
  let q=quizData[current];
  let html=`<div class='q-box'><h3>${current+1}. ${q.q}</h3>`;
  q.o.forEach((opt,i)=>{
    html+=`<div class='option' onclick='select(${i})'>${opt}</div>`;
  });
  html+="</div>";
  document.getElementById("quiz").innerHTML=html;
  document.getElementById("progress").style.width=((current+1)/quizData.length*100)+"%";
}

function select(i){
  answers[current]=i;
  loadQ();
}

function nextQ(){ if(current<quizData.length-1){current++; loadQ();}}
function prevQ(){ if(current>0){current--; loadQ();}}

function submitQuiz(){
  let score=0;
  quizData.forEach((q,i)=>{
    if(answers[i]==q.a) score++;
  });
  document.getElementById("result").innerHTML=
    `🎯 Score: ${score}/30 <br> 📊 ${Math.round(score/30*100)}%`;
}

loadQ();

let time=900;
setInterval(()=>{
  let m=Math.floor(time/60);
  let s=time%60;
  document.getElementById("timer").innerText=m+":"+(s<10?"0"+s:s);
  time--;
  if(time<=0) submitQuiz();
},1000);
