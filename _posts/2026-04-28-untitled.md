<script>
  function selectOption(element, selectedIndex) {
    const question = element.closest('.quiz-question');
    const options = question.querySelectorAll('.quiz-option');

    if (question.classList.contains('answered')) return;

    options.forEach((option) => option.classList.remove('selected'));
    element.classList.add('selected');
    question.setAttribute('data-selected', selectedIndex);
  }

  function submitQuiz() {
    const questions = document.querySelectorAll('.quiz-question');
    let score = 0;

    questions.forEach((question) => {
      const correctAnswer = Number(question.getAttribute('data-answer'));
      const selectedAnswer = Number(question.getAttribute('data-selected'));
      const options = question.querySelectorAll('.quiz-option');

      question.classList.remove('correct', 'wrong');
      question.classList.add('answered');

      options.forEach((option, index) => {
        option.disabled = true;
        option.classList.remove('correct-answer', 'wrong-answer');

        if (index === correctAnswer) {
          option.classList.add('correct-answer');
        }

        if (index === selectedAnswer && selectedAnswer !== correctAnswer) {
          option.classList.add('wrong-answer');
        }
      });

      if (!Number.isNaN(selectedAnswer) && selectedAnswer === correctAnswer) {
        score += 1;
        question.classList.add('correct');
      } else {
        question.classList.add('wrong');
      }
    });

    const scoreCard = document.getElementById('scoreCard');
    const scoreNum = document.getElementById('scoreNum');
    const scoreLabel = document.getElementById('scoreLabel');

    let feedback = 'আরও অনুশীলন করো।';
    if (score === 10) feedback = 'অসাধারণ! সবগুলো উত্তর সঠিক হয়েছে।';
    else if (score >= 8) feedback = 'খুব ভালো! বিষয়টি ভালোভাবে আয়ত্ত হয়েছে।';
    else if (score >= 5) feedback = 'ভালো চেষ্টা। আরও একটু ঝালাই করলে আরও ভালো হবে।';

    scoreNum.textContent = `${score}/10`;
    scoreLabel.textContent = feedback;
    scoreCard.classList.add('show');
    scoreCard.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
  }

  function resetQuiz() {
    const questions = document.querySelectorAll('.quiz-question');

    questions.forEach((question) => {
      question.classList.remove('correct', 'wrong', 'answered');
      question.removeAttribute('data-selected');

      question.querySelectorAll('.quiz-option').forEach((option) => {
        option.disabled = false;
        option.classList.remove('selected', 'correct-answer', 'wrong-answer');
      });
    });

    document.getElementById('scoreCard').classList.remove('show');
    document.getElementById('scoreNum').textContent = '0/10';
    document.getElementById('scoreLabel').textContent = '';
  }
</script>
