<script>
//<![CDATA[
let time=900;
const timerDisplay = document.getElementById('timer');

let countdown = setInterval(()=>{
  let m=Math.floor(time/60), s=time%60;
  // innerHTML fixes the entity/emoji display issue
  timerDisplay.innerHTML=`⏱️ ${m}:${s<10?'0'+s:s}`;
  time--;
  if(time<0) { clearInterval(countdown); document.getElementById('submit').click(); }
},1000);

const quiz=document.querySelector('.quiz');
const bar=document.getElementById('progressBar');

quiz.addEventListener('click',e=>{
  if(!e.target.classList.contains('opt'))return;
  const q=e.target.closest('.q');
  if(q.classList.contains('done'))return;

  q.querySelectorAll('.opt').forEach(o=>o.classList.remove('selected'));
  e.target.classList.add('selected');
  q.dataset.sel=[...q.children].indexOf(e.target)-1;

  let ans=document.querySelectorAll('.q[data-sel]').length;
  bar.style.width=(ans/15)*100+'%';
});

document.getElementById('submit').onclick=()=>{
  clearInterval(countdown);
  let score=0;
  document.querySelectorAll('.q').forEach(q=>{
    let c=q.dataset.a, s=q.dataset.sel;
    if(s==c){score++; q.classList.add('correct')}
    else q.classList.add('wrong');
    q.classList.add('done');
    
    // Highlight correct answer if missed
    let opts = q.querySelectorAll('.opt');
    if(opts[c]) opts[c].classList.add('correct');
    if(s !== null && s !== c && opts[s]) opts[s].classList.add('wrong');
  });

  let p=(score/15)*100;
  let g=p>=80?"A+":p>=70?"A":p>=60?"B":p>=50?"C":"Fail";

  let sc=document.getElementById('score');
  sc.innerHTML=`🎯 Score: ${score}/15 <br> 📊 ${p.toFixed(1)}% | Grade: ${g}`;
  sc.classList.add('show');
  window.scrollTo({ top: document.body.scrollHeight, behavior: 'smooth' });
};
//]]>
</script>
