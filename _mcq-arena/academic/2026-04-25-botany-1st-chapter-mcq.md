<script>
const quiz = document.querySelector('.quiz');
const progressBar = document.getElementById('progressBar');
const progressText = document.getElementById('progressText');

function updateProgress(){
  const qs = document.querySelectorAll('.q');
  let answered = 0;

  qs.forEach(q=>{
    if(q.dataset.sel !== undefined) answered++;
  });

  const percent = (answered / qs.length) * 100;
  progressBar.style.width = percent + '%';
  progressText.innerText = `${answered} / ${qs.length} answered`;
}

quiz.addEventListener('click', e=>{
  if(!e.target.classList.contains('opt')) return;

  const q = e.target.closest('.q');
  if(q.classList.contains('done')) return;

  const allQ = [...document.querySelectorAll('.q')];
  const currentIndex = allQ.indexOf(q);

  q.querySelectorAll('.opt').forEach(o=>o.classList.remove('selected'));
  e.target.classList.add('selected');

  q.dataset.sel = [...q.querySelectorAll('.opt')].indexOf(e.target);

  updateProgress();

  const nextQ = allQ[currentIndex + 1];
  if(nextQ){
    setTimeout(()=>{
      nextQ.scrollIntoView({behavior:'smooth',block:'center'});
    },250);
  }
});

document.getElementById('submit').onclick=()=>{
  let score=0;
  const qs=document.querySelectorAll('.q');

  qs.forEach(q=>{
    const c=q.dataset.a;
    const s=q.dataset.sel;

    if(s!==undefined){
      const opts=q.querySelectorAll('.opt');

      if(s==c){score++;q.classList.add('correct')}
      else q.classList.add('wrong');

      opts[c].classList.add('correct');
      if(s!=c) opts[s].classList.add('wrong');
    }

    q.classList.add('done');
  });

  progressBar.style.width='100%';

  const sc=document.getElementById('score');
  sc.innerText=`Score: ${score}/${qs.length}`;
  sc.classList.add('show');
};
</script>
