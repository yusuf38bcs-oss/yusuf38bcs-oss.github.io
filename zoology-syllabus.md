<script>
  (function() {
    var panels = document.querySelectorAll(".syllabus-module-panel");
    panels.forEach(function(panel) {
      panel.addEventListener("toggle", function() {
        if (panel.open) {
          panels.forEach(function(otherPanel) {
            if (otherPanel !== panel && otherPanel.open) {
              otherPanel.open = false;
            }
          });
        }
      });
    });
  })();
</script>
